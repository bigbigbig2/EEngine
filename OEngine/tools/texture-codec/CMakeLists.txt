cmake_minimum_required(VERSION 3.23)
project(EEngineTextureCodec C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_TESTS OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_TOOLS OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_GL_UPLOAD OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_VK_UPLOAD OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_ETC_UNPACK OFF CACHE BOOL "" FORCE)
set(KTX_FEATURE_KTX1 OFF CACHE BOOL "" FORCE)
set(BASISU_SUPPORT_SSE OFF CACHE BOOL "" FORCE)
add_subdirectory(${KTX_SOURCE} ktx EXCLUDE_FROM_ALL)
add_executable(ktx_pc KtxRead.cpp)
target_link_libraries(ktx_pc PRIVATE ktx_read)
target_include_directories(ktx_pc PRIVATE ${KTX_SOURCE}/include ${KTX_SOURCE}/lib ${KTX_SOURCE}/external/dfdutils)
target_link_options(ktx_pc PRIVATE --bind -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createKtxPcModule
    -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=100663296 -sABORTING_MALLOC=0 -sINITIAL_MEMORY=16777216 -sSTACK_SIZE=1048576
    -sENVIRONMENT=web,worker,node -sEXPORTED_RUNTIME_METHODS=HEAPU8 -sDISABLE_EXCEPTION_CATCHING=0)
target_compile_options(ktx_pc PRIVATE -fexceptions)
