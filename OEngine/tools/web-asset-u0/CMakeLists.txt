cmake_minimum_required(VERSION 3.20)
project(eengine_web_asset_u0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BASISU_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BASISU_SSE OFF CACHE BOOL "" FORCE)
set(BASISU_OPENCL OFF CACHE BOOL "" FORCE)
set(BASISU_DISABLE_ANDROID_ASTC_DECOMP ON CACHE BOOL "" FORCE)
add_subdirectory("${BASIS_SOURCE}" basis EXCLUDE_FROM_ALL)
add_executable(xubc7-probe Xubc7Encode.cpp Xubc7Probe.cpp)
target_include_directories(xubc7-probe PRIVATE "${BASIS_SOURCE}/encoder" "${BASIS_SOURCE}/transcoder" "${BASIS_SOURCE}/zstd")
target_compile_definitions(xubc7-probe PRIVATE BASISU_SUPPORT_SSE=0 BASISD_SUPPORT_KTX2_ZSTD=1)
target_link_libraries(xubc7-probe PRIVATE basisu_encoder)
